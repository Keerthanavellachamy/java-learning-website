Connect Java with MySQL

The following class creates a connection between Java and the MySQL database.

DBConnection.java 
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.SQLException;

public class DBConnection {

    private static final String URL =
        "jdbc:mysql://localhost:3306/college_db";

    private static final String USER = "root";

    private static final String PASSWORD =
        "your_password";

    public static Connection getConnection()
            throws SQLException {

        return DriverManager.getConnection(
            URL,
            USER,
            PASSWORD
        );
    }
}
⚠️ Replace your_password with the password you created for your MySQL root user.

Add Student

Use PreparedStatement to safely insert student information into the database.

Add Student 
String sql =
    "INSERT INTO students (name, department, year, email) " +
    "VALUES (?, ?, ?, ?)";

PreparedStatement ps =
    connection.prepareStatement(sql);

ps.setString(1, name);
ps.setString(2, department);
ps.setInt(3, year);
ps.setString(4, email);

ps.executeUpdate();
The ? symbols are placeholders. Values are added using setString() and setInt().

Complete Student Management System

Now combine the database connection, CRUD operations and menu into one complete Java application.

StudentManagement.java 
import java.sql.*;
import java.util.Scanner;

public class StudentManagement {

    static final String URL =
        "jdbc:mysql://localhost:3306/college_db";

    static final String USER = "root";

    static final String PASSWORD =
        "your_password";

    static Scanner scanner =
        new Scanner(System.in);

    public static void main(String[] args) {

        while (true) {

            System.out.println(
                "\n--- Student Management System ---"
            );

            System.out.println("1. Add Student");
            System.out.println("2. View Students");
            System.out.println("3. Update Student");
            System.out.println("4. Delete Student");
            System.out.println("5. Exit");

            System.out.print("Enter choice: ");

            int choice = scanner.nextInt();
            scanner.nextLine();

            switch (choice) {

                case 1:
                    addStudent();
                    break;

                case 2:
                    viewStudents();
                    break;

                case 3:
                    updateStudent();
                    break;

                case 4:
                    deleteStudent();
                    break;

                case 5:
                    System.out.println("Thank you!");
                    return;

                default:
                    System.out.println("Invalid choice");
            }
        }
    }

    static Connection connect()
            throws SQLException {

        return DriverManager.getConnection(
            URL,
            USER,
            PASSWORD
        );
    }

    static void addStudent() {

        try (Connection connection = connect()) {

            String sql =
                "INSERT INTO students " +
                "(name, department, year, email) " +
                "VALUES (?, ?, ?, ?)";

            PreparedStatement ps =
                connection.prepareStatement(sql);

            System.out.print("Enter name: ");
            String name = scanner.nextLine();

            System.out.print("Enter department: ");
            String department = scanner.nextLine();

            System.out.print("Enter year: ");
            int year = scanner.nextInt();
            scanner.nextLine();

            System.out.print("Enter email: ");
            String email = scanner.nextLine();

            ps.setString(1, name);
            ps.setString(2, department);
            ps.setInt(3, year);
            ps.setString(4, email);

            ps.executeUpdate();

            System.out.println(
                "Student added successfully!"
            );

        } catch (SQLException e) {
            e.printStackTrace();
        }
    }

    static void viewStudents() {

        try (Connection connection = connect()) {

            String sql =
                "SELECT * FROM students";

            Statement statement =
                connection.createStatement();

            ResultSet result =
                statement.executeQuery(sql);

            System.out.println(
                "\nID | Name | Department | Year | Email"
            );

            while (result.next()) {

                System.out.println(
                    result.getInt("id") + " | " +
                    result.getString("name") + " | " +
                    result.getString("department") + " | " +
                    result.getInt("year") + " | " +
                    result.getString("email")
                );
            }

        } catch (SQLException e) {
            e.printStackTrace();
        }
    }

    static void updateStudent() {

        try (Connection connection = connect()) {

            System.out.print("Enter student ID: ");
            int id = scanner.nextInt();
            scanner.nextLine();

            System.out.print(
                "Enter new department: "
            );

            String department = scanner.nextLine();

            System.out.print("Enter new year: ");
            int year = scanner.nextInt();

            String sql =
                "UPDATE students " +
                "SET department = ?, year = ? " +
                "WHERE id = ?";

            PreparedStatement ps =
                connection.prepareStatement(sql);

            ps.setString(1, department);
            ps.setInt(2, year);
            ps.setInt(3, id);

            int rows =
                ps.executeUpdate();

            if (rows > 0) {
                System.out.println(
                    "Student updated!"
                );
            } else {
                System.out.println(
                    "Student not found."
                );
            }

        } catch (SQLException e) {
            e.printStackTrace();
        }
    }

    static void deleteStudent() {

        try (Connection connection = connect()) {

            System.out.print("Enter student ID: ");
            int id = scanner.nextInt();

            String sql =
                "DELETE FROM students WHERE id = ?";

            PreparedStatement ps =
                connection.prepareStatement(sql);

            ps.setInt(1, id);

            int rows =
                ps.executeUpdate();

            if (rows > 0) {
                System.out.println(
                    "Student deleted!"
                );
            } else {
                System.out.println(
                    "Student not found."
                );
            }

        } catch (SQLException e) {
            e.printStackTrace();
        }
    }
}

Try-With-Resources

Java 
try (Connection connection = connect()) {

    // Database operations

}
💡 Try-with-resources automatically closes the database connection after the block finishes. This helps prevent resource leaks.

Common Errors

Unknown Database

Check whether college_db has been created in MySQL.

Access Denied

Check your MySQL username and password.

No Suitable Driver

Make sure MySQL Connector/J is added to your Java project.

Table Doesn't Exist

Make sure the students table was created.

Connection Failure

Check that MySQL Server is running and the port is correct.
